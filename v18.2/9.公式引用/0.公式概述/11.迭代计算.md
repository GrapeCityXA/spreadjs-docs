# calculating-iterative

SpreadJS 提供了对迭代计算的支持，这也被称为循环引用，就像在 Excel 中一样。一般来说，迭代计算是对工作表进行反复计算，直到满足确定的数值条件为止。当工作表中某个单元格的公式直接或间接地引用了该单元格自身时，就会使用迭代计算来重新计算该公式，直到找到特定的结果。
你可以通过使用 `iterativeCalculation` 属性在工作表中启用迭代计算。你还可以通过设置 `iterativeCalculationMaximumIterations` 属性来指定公式应重新计算的次数。此外，你也可以通过设置 `iterativeCalculationMaximumChange` 属性来指定两次计算值之间的最大变化量。
启用迭代计算后，所有的循环引用都会进行迭代计算，直到所有的值变化量小于 `IterativeCalculationMaximumChange` 或者迭代次数达到 `IterativeCalculationMaximumIterations` 为止。但是，当禁用迭代计算时，所有循环引用的单元格值将变为零，并且被它引用的其他单元格也将变为零。
## 应用场景
你可以在以下应用场景中使用迭代计算：
**计算客户未来的投资价值**
假设一位客户有 50,000 美元，并且想将这笔钱投资到现金存款（CD）账户中，该账户的月利率为 4.75%。为了计算 2 年后的总价值，我们将计算每个月的本金，并将利息加到上个月的本金上。在这里，我们将 `iterativeCalculation` 属性设置为 `True` 以启用迭代计算，并将最大迭代次数设置为 24，以便计算 24 个月（2 年）后的利息，即第 25 个月末的现金总值。
![](/DOCUMENT_SITE_LINK_PREFIX_HERE/document-site-files/images/5fefd5d8-238b-4c13-893b-6f22fcc5b9dd/investment-iterative.d25793.png)
可以使用下面的代码示例来描述这个应用场景：
```javascript
$(document).ready(function () {
    // 初始化 Spread
    var spread = new GC.Spread.Sheets.Workbook(document.getElementById('ss'), { sheetCount: 1 });
    // 获取活动工作表
    var sheet = spread.getSheet(0);
    // 设置列宽
    sheet.setColumnWidth(0, 30);
    sheet.setColumnWidth(1, 110);
    sheet.setColumnWidth(2, 130);
    sheet.setColumnWidth(3, 30);
    sheet.setColumnWidth(4, 30);
    sheet.setColumnWidth(5, 230);
    sheet.setColumnWidth(6, 130);
    // 启用迭代计算
    spread.options.iterativeCalculation = true;
    spread.options.iterativeCalculationMaximumIterations = 24;
    spread.options.iterativeCalculationMaximumChange = 0.1;
    // 设置文本和值
    sheet.setText(1, 1, "初始现金");
    sheet.setValue(1, 2, 50000);
    sheet.setText(2, 1, "利率");
    sheet.setValue(2, 2, 0.0475);
    sheet.setText(4, 1, "月份/年份");
    sheet.setText(4, 2, "现金总值");
    /
    / 创建数组
    var items = [
    "2019 年 1 月", "2019 年 2 月", "2019 年 3 月", "2019 年 4 月", "2019 年 5 月", "2019 年 6 月", "2019 年 7 月", "2019 年 8 月", "2019 年 9 月", "2019 年 10 月", "2019 年 11 月", "2019 年 12 月",
    "2020 年 1 月", "2020 年 2 月", "2020 年 3 月", "2020 年 4 月", "2020 年 5 月", "2020 年 6 月", "2020 年 7 月", "2020 年 8 月", "2020 年 9 月", "2020 年 10 月", "2020 年 11 月", "2020 年 12 月", "2021 年 1 月"
    ];
    // 设置数组
    sheet.setArray(5, 1, items);
    // 设置公式
    sheet.setFormula(5, 2, '=C2*(1+C3)');
    sheet.setFormula(6, 2, '=C6*(1+C3)');
    for (var i = 6; i < 30; i++) {
        sheet.setFormula(i, 2, '=C' + i + '*(1+C3)');
    }
    // 创建表格以在一个区域上添加样式
    sheet.tables.add("Table1", 4, 1, 26, 2, GC.Spread.Sheets.Tables.TableThemes.medium6);
    // 设置值和公式
    sheet.setText(1, 5, "初始现金");
    sheet.setValue(1, 6, 50000);
    sheet.setText(2, 5, "利率");
    sheet.setValue(2, 6, 0.0475);
    sheet.setText(3, 5, "2 年后的现金总值（截至 2021 年 1 月）");
    sheet.setFormula(3, 6, '=G2*(1+G3)');
    // 此公式将生成循环引用
    sheet.setFormula(1, 6, '=G4');
    // 设置格式
    sheet.setFormatter(1, 2, '$#,##0.00');
    sheet.setFormatter(2, 2, '0.00%');
    sheet.setFormatter(1, 6, '$#,##0.00');
    sheet.setFormatter(2, 6, '0.00%');
    sheet.setFormatter(3, 6, '$#,##0.00');
    sheet.setFormatter(5, 2, '$#,##0.00');
    for (var i = 6; i < 30; i++) {
        sheet.setFormatter(i, 2, '$#,##0.00');
    }
    // 为单元格区域设置边框
    sheet.getRange("B2:C3", GC.Spread.Sheets.SheetArea.viewport).setBorder(new GC.Spread.Sheets.LineBorder("black", GC.Spread.Sheets.LineStyle.thin), { all: true }, 3);
    sheet.getRange("F2:G4", GC.Spread.Sheets.SheetArea.viewport).setBorder(new GC.Spread.Sheets.LineBorder("black", GC.Spread.Sheets.LineStyle.thin), { all: true }, 3);
    sheet.getRange("B5:C30", GC.Spread.Sheets.SheetArea.viewport).setBorder(new GC.Spread.Sheets.LineBorder("black", GC.Spread.Sheets.LineStyle.thin), { all: true }, 3);
    // 为单元格区域设置背景颜色
    sheet.getRange("C2:C3").backColor("#D9E3F2");
    sheet.getRange("G2:G4").backColor("#D9E3F2");
});
```
**为单元格添加时间戳**
迭代计算可用于在工作表中设置时间戳。例如，你正在管理一个任务列表，并且想在每个任务后的相邻单元格中添加日期和时间。为了在时间戳应用场景中启用迭代计算，我们将 `iterativeCalculation` 属性设置为 `true`。在这里，我们使用了 `iterativeCalculationMaximumIterations` 属性并将其设置为 24，以便告知 SpreadJS 运行公式的最大次数。通过迭代计算，每次我们在“员工姓名”列的新行中输入一个新项时，“时间戳”列中的时间戳将自动更新，直到达到第 23 行。
![](/DOCUMENT_SITE_LINK_PREFIX_HERE/document-site-files/images/5fefd5d8-238b-4c13-893b-6f22fcc5b9dd/timestampiterative.d4e87c.gif)
可以使用下面的代码示例来描述这个应用场景：
```javascript
$(document).ready(function () {
    // 初始化 Spread
    var spread = new GC.Spread.Sheets.Workbook(document.getElementById('ss'), { sheetCount: 1 });
    // 获取活动工作表
    var sheet = spread.getSheet(0);
    sheet.suspendPaint();
    // 设置列宽
    sheet.setColumnWidth(0, 200);
    sheet.setColumnWidth(1, 200);
    // 启用迭代计算
    spread.options.iterativeCalculation = true;
    spread.options.iterativeCalculationMaximumIterations = 24;
    spread.options.iterativeCalculationMaximumChange = 0.1;
    // 为单元格区域设置背景颜色
    sheet.getRange("A1:B1").backColor("#D9E3F2");
    // 在单元格中设置文本
    sheet.setText(0, 0, "员工姓名");
    sheet.setText(0, 1, "时间戳");
    for (var i = 2; i < spread.options.iterativeCalculationMaximumIterations; i++) {
        sheet.setFormula(i - 1, 1, '=IF(A' + i + '<>"",IF(B' + i + '<>"",B' + i + ',NOW()),"")');
        sheet.setFormatter(i - 1, 1, 'dd-mm-yyyy hh:mm:ss');
    }
    // 示例数据
    sheet.setText(1, 0, "安迪");
    var style = new GC.Spread.Sheets.Style();
    style.font = "bold 12px Arial";
    style.foreColor = "white";
    style.backColor = "#5B9BD5";
    style.hAlign = GC.Spread.Sheets.HorizontalAlign.center;
    style.vAlign = GC.Spread.Sheets.VerticalAlign.center;
    for (var i = 0; i < 2; i++)
        sheet.setStyle(0, i, style, GC.Spread.Sheets.SheetArea.viewport);
    sheet.resumePaint();
});
```
