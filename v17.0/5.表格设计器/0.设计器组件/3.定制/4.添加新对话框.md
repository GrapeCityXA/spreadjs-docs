# add_new_dialog

在上下文菜单组件中选择“Signature Dialog”选项时，按照以下步骤添加“Insert Signature”对话框。

- 打开并编辑config.js文件，在其中添加“Insert Signature"”选项“contextMenu”标签。
```JavaScript
var config = {
    // 其他项目内容
    "contextMenu":[
            
           "designer.insertSignature",
            // 其他上下文菜单项

            ],
    ...
    }
```


- 在项目中设置命令为commandMap。
```JavaScript
config.commandMap = {
                "designer.insertSignature": {
                    text: "Insert Signature",
                    commandName: "designer.insertSignature",
                    visibleContext: "ClickRowHeader",
                    execute:
                    // execute_InsertSignature，后面是一个简单的演示代码片段
                    () =>{
                        console.log("Insert Signature");
                    }
                }
            }
```


- 通过传递以下参数初始化设计器实例可自定义配置的"config"参数。
```JavaScript
// 初始化设计器实例
var spread = new GC.Spread.Sheets.Workbook(document.getElementById("ss"));
var designer = new GC.Spread.Sheets.Designer.Designer(document.getElementById("gc-designer-container"), config, spread);
```


- 打开并编辑config.js文件，在单击上下文菜单中的“Insert Signature”项时添加新的对话框“Signature”对话框
```JavaScript
// 点击上下文菜单中的“Insert Signature”项，初始化一个对话框模板实例
let insertSignature = {
                templateName: "insertSignature",
                "title": "Signature Dialog",
                "content": [
                    {
                        "type": "FlexContainer",
                        "children": [
                            {
                                "type": "TextBlock",
                                "margin": "5px -4px",
                                "text": "Insert Range"
                            },
                            {
                                "type": "RangeSelect",
                                "title": "Select Range",
                                "absoluteReference": true,
                                "needSheetName": false,
                                "margin": "5px -5px"
                            }
                        ]
                    }
                ]
            }
```
