# SetChartColorwithTransparency

SpreadJS 在配置不同图表元素时支持透明度设置，允许用户自定义图表区域背景颜色、图表数据标签、图表坐标轴、系列等。
下图展示了一个描绘四个不同国家销售数据的图表，其中不同的图表元素已进行自定义以设置透明度。
![chart_transparency](/DOCUMENT_SITE_LINK_PREFIX_HERE/document-site-files/images/5fefd5d8-238b-4c13-893b-6f22fcc5b9dd/chart_transparency.13b309.png)
以下代码示例展示了如何设置带有透明度的图表颜色。

```javascript
window.onload = function () {
    var spread = new GC.Spread.Sheets.Workbook(document.getElementById('ss'));
    sheet = spread.getActiveSheet();
    sheet.suspendPaint();
    var dataArray = [
        ['', '1月', '2月', '3月', '4月', '5月', '6月', '7月', '8月', '9月', '10月', '11月', '12月'],
        ['东京', 49.9, 71.5, 106.4, 129.2, 144.0, 176.0, 135.6, 148.5, 216.4, 194.1, 95.6, 54.4],
        ['纽约', 83.6, 78.8, 98.5, 93.4, 106.0, 84.5, 105.0, 104.3, 91.2, 83.5, 106.6, 92.3],
        ['伦敦', 48.9, 38.8, 39.3, 41.4, 47.0, 48.3, 59.0, 59.6, 52.4, 65.2, 59.3, 51.2],
        ['柏林', 42.4, 33.2, 34.5, 39.7, 52.6, 75.5, 57.4, 60.4, 47.6, 39.1, 46.8, 51.1],
    ];

    sheet.setArray(0, 0, dataArray);
    // 添加一个簇状柱形图类型的柱形图
    chart_columnClustered = sheet.charts.add('chart_columnClustered', GC.Spread.Sheets.Charts.ChartType.columnClustered, 300, 180, 600, 400, 'A1:M5');
    // 设置图表标题的透明度
    chart_columnClustered.title({
        text: '销售数据',
        color: '蓝色',
        transparency: 0.5,
    });

    // 设置图表图例背景颜色和边框的透明度
    chart_columnClustered.legend({
        backColor: '红色',
        backColorTransparency: 0.5,
        borderStyle: {
            color: '绿色',
            width: 5,
            transparency: 0.7,
        },
    });

    // 设置图表区域背景颜色的透明度
    chart_columnClustered.chartArea({
        backColor: '红色',
        backColorTransparency: 0.9,
        color: '黑色',
        transparency: 0.6,
    });

    // 设置图表数据标签的透明度
    chart_columnClustered.dataLabels({
        showValue: true,
        color: '红色',
        transparency: 0.6,
    });
    // 设置图表坐标轴的透明度
    chart_columnClustered.axes({
        primaryCategory: {
            lineStyle: {
                color: '蓝色',
                width: 5,
                transparency: 0.8,
            },
            style: {
                color: '黑色',
                transparency: 0.7,
            },
            title: {
                color: '深色',
                transparency: 0.4,
            },
        },
    });
    // 设置图表系列的透明度
    var series1 = chart_columnClustered.series().get(0);
    series1.backColor = '红色';
    series1.backColorTransparency = 0.5;
    series1.border = {
        color: '蓝色',
        width: 4,
        transparency: 0.6,
    };
    chart_columnClustered.series().set(0, series1);
    sheet.resumePaint();
};
```
