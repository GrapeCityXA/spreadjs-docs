# dynamic-column-addition

> type=warning
> AlexZ: 重新整理背景说明

### 需求：在新增列之后，能够根据绑定的数据更新自动整个表格

***

#### 背景：

用户希望能够在给表格新增列之后，之前绑定的数据仍然能够自动绑定。
对于这种动态实现列添加的需求，通过分析核心代码逻辑就是通过[单元格类型](https://demo.grapecity.com.cn/spreadjs/SpreadJSTutorial/features/cells/cell-types/basic/purejs)一个按钮类型的列添加按钮，并监听这个按钮的点击
因此按照我们既定的想法我们做以下代码的设计

#### 核心代码：

1\. 新增按钮的逻辑编写

```javascript
// 设定一个按钮点击事件用于添加表格列，原理就是，添加一列，添加绑定路径，重新设定绑定数据
workbook.bind(GC.Spread.Sheets.Events.ButtonClicked, function (s, e) {
    let sheet = e.sheet, row = e.row, col = e.col;
    let cell = sheet.getCell(row, col);
    if (cell.value() === "addColumn") {
        let newColumnValue = sheet.getValue(row, col - 1);//同一行前一列
        if (newColumnValue) {
            table.insertColumns(columnsInfo.length - 1, 1, true)// 在该表中的指定表格列索引之后插入计数编号列
            var tableColumn = new GC.Spread.Sheets.Tables.TableColumn();
            tableColumn.name(newColumnValue.toUpperCase());
            tableColumn.dataField(newColumnValue);// 就是value
            columnsInfo.push(tableColumn)// 动态数组记录信息
            table.bindColumns(columnsInfo);// 重新设置绑定信息
            var dataSource = new GC.Spread.Sheets.Bindings.CellBindingSource(data);// 重新绑定数据
            sheet.setDataSource(dataSource)
        }
    }
})
```

2\. 表格结构变化监听的细节编写

```javascript
workbook.bind(GC.Spread.Sheets.Events.ValueChanged, function (s, e) {// 表头列数据改变
    let sheet = e.sheet, row = e.row, col = e.col;
    let cell = sheet.getCell(row, col), value = cell.value();
    if (value == "" || value == null) {
        return;
    }
    var table = sheet.tables.find(row, col);// 获取table
    if (table) {
        var range = table.range();
        if (col === range.col) {// 如果是表头列数据改变
            if (lineList.indexOf(value) >= 0) {
                sheet.setValue(row, col + 1, value + "B");
                sheet.setValue(row, col + 2, value + "C");
            }
        }
    }
});

workbook.bind(GC.Spread.Sheets.Events.TableRowsChanged, function (s, e) {// 表格行添加事件，目的是为了表头列的设置
    if (e.propertyName = "tableInsertRows") {// 表格行改变的具体操作是插入了一行
        var sheet = e.sheet, table = e.table;
        updateTable(sheet, table);
    }
});

workbook.bind(GC.Spread.Sheets.Events.DragDropBlock, function (s, e) {// 拖放单元格事件，目的是在表格区域中单个单元格被拖动时
    debugger
    if (!(e.rowCount === 1 && e.colCount === 1)) {
        return;
    }
    let sheet = e.sheet, fromRow = e.fromRow, fromCol = e.fromCol;
    var table = sheet.tables.find(fromRow, fromCol);
    if (table) {
        var range = table.range();
        if (fromRow !== range.row) {// 如果拖动位置的行不是表头就算了，如果是表头，那就有的说了
            return;
        }
        e.cancel = true;// 并且拖动动作取消
        setTimeout(function () {
            sheet.setValue(fromRow, fromCol, sheet.getValue(fromRow, fromCol) + "2")
            //your changed columnInfo code
        }, 10)
    }
});
```
