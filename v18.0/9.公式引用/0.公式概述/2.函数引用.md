# function-reference

函数引用或 eta-reduced lambda 表达式通过在函数对其参数没有进一步操作时消除不必要的抽象来简化函数。这使得在可能的情况下，使用缩写语法。
SpreadJS 允许你在自定义名称、LET 函数的“name\_value”参数、LAMBDA 函数中使用函数引用，并在 MAP、BYCOL、BYROW、MAKEARRAY、REDUCE、SCAN 和 GROUPBY 函数中作为 LAMBDA 函数参数使用。它还允许你在公式函数中引用自定义函数。

## 自定义名称中的函数引用

你可以定义一个内容为函数名的自定义名称，并在公式中使用它。

```javascript
// 向单元格添加值。
sheet.setValue(0, 2, 10);
sheet.setValue(1, 2, 20);
sheet.setValue(2, 2, -5);
sheet.setValue(3, 2, 0);
sheet.setValue(4, 2, 20);
sheet.setValue(0, 0, "公式结果: ->");
sheet.setValue(1, 0, "公式文本: ->");

// 设置单元格 B1 的颜色。
sheet.getCell(0, 1).backColor('黄色');
sheet.setValue(7, 0, "添加了内容为 'SUM' 函数名的自定义名称 'op'。");

// 向单元格 B2 添加 FORMULATEXT 函数。
sheet.setFormula(1, 1, "=FORMULATEXT(B1)");

// 添加内容为函数名的自定义名称。
spread.addCustomName("op", "SUM");

// 将自定义名称用作公式。
sheet.setFormula(0, 1, "op(C1:C5)");
```

## LET 函数中的函数引用

你可以将 LET 函数的“name\_value”参数定义为一个函数，并用于计算。

```javascript
// 向单元格添加值。
sheet2.setValue(0, 2, 10);
sheet2.setValue(1, 2, 20);
sheet2.setValue(2, 2, -5);
sheet2.setValue(3, 2, 0);
sheet2.setValue(4, 2, 20);
sheet2.setValue(0, 0, "公式结果: ->");
sheet2.setValue(1, 0, "公式文本: ->");

// 设置单元格 B1 的颜色。
sheet2.getCell(0, 1).backColor('黄色');
sheet2.setValue(7, 0, "在 name_value 中定义了 'SUM' 函数，并在 'LET' 函数中使用");

// 向单元格 B2 添加 FORMULATEXT 函数。
sheet2.setFormula(1, 1, "=FORMULATEXT(B1)");

// 将 “name_value” 设为 SUM 函数并用于计算。
sheet2.setFormula(0, 1, "=LET(op,SUM,op(C1:C5))");
```

## LAMBDA 函数参数中的函数引用

你可以将函数名作为 LAMBDA 函数中的参数使用，并用于计算。你还可以在 MAP、BYCOL、BYROW、MAKEARRAY、REDUCE、SCAN 和 GROUPBY 函数中将函数名作为 LAMBDA 函数参数使用。

```javascript
// LAMBDA 函数
// 向单元格添加值。
sheet3.setValue(0, 0, "公式结果: ->");
sheet3.setValue(1, 0, "公式文本: ->");

// 设置单元格 B1 的颜色。
sheet3.getCell(0, 1).backColor('黄色');

// 添加 LAMBDA 函数。
sheet3.setFormula(0, 1, "=LAMBDA(op,op(123,231))(SUM)");

// 向单元格 B2 添加 FORMULATEXT 函数。
sheet3.setFormula(1, 1, "=FORMULATEXT(B1)");


// MAP 函数
// 向单元格添加值。
sheet3.setValue(5, 0, "公式结果: ->");
sheet3.setValue(4, 0, "公式文本: ->");

// 设置单元格 B6:C9 的颜色。
sheet3.getRange("B6:C9").backColor('浅蓝色');

// 设置单元格 B6 的颜色。
sheet3.getCell(5, 1).backColor('黄色');

// 向工作表添加表格。
sheet3.tables.add('table1', 4, 4, 5, 2);

// 向表格添加值。
sheet3.setArray(4, 4, [
    ['Col1', 'Col2'],
    [20.33, -19],
    [-20.3, 0],
    [-12, -15.3],
    [17, 2]
]);

// 向单元格 B5 添加 FORMULATEXT 函数。
sheet3.setFormula(4, 1, "=FORMULATEXT(B6)");

// 添加 MAP 函数，将函数名作为 LAMBDA 函数参数。
sheet3.setFormula(5, 1, "=MAP(E6:F9,ABS)");


// BYCOL 函数
// 向单元格添加值。
sheet3.setValue(10, 0, "公式文本: ->");
sheet3.setValue(11, 0, "公式结果: ->");

// 设置单元格 B12:F12 的颜色。
sheet3.getRange("B12:F12").backColor('浅蓝色');

// 设置单元格 B12 的颜色。
sheet3.getCell(11, 1).backColor('黄色');

// 向工作表添加表格。
sheet3.tables.add('table2', 10, 7, 3, 5);

// 向表格添加值。
sheet3.setArray(10, 7, [
    ['Col1', 'Col2', 'Col3', 'Col4', 'Col5'],
    [1, 2, 9, 10, -5],
    [20, 0, 5, -3, 30]
]);

// 向单元格 B11 添加 FORMULATEXT 函数。
sheet3.setFormula(10, 1, "=FORMULATEXT(B12)");

// 添加 BYCOL 函数，将函数名作为 LAMBDA 函数参数。
sheet3.setFormula(11, 1, "=BYCOL(H12:L13,MAX)");


// GROUPBY 函数
// 向单元格添加值。
sheet3.setValue(15, 0, "公式文本: ->");
sheet3.setValue(16, 0, "公式结果: ->");

// 设置单元格 B17:D29 的颜色。
sheet3.getRange("B17:D29").backColor('浅蓝色');

// 设置单元格 B17 的颜色。
sheet3.getCell(16, 1).backColor('黄色');

// 向工作表添加表格。
sheet3.tables.add('table3', 15, 7, 13, 7);

// 向表格添加值。
sheet3.setArray(15, 7, [
    ['年份', '类别', '产品', '状态', '销售额', '评分', '利润'],
    [2023, '电子产品', '智能电视', '在售', 15000, 4.5, 250],
    [2023, '时尚', '设计师牛仔裤', '停产', 8000, 4.7, 150],
    [2024, '食品', '有机格兰诺拉麦片', '在售', 5000, 4.8, 100],
    [2023, '书籍', '畅销小说', '在售', 12000, 4.6, 180],
    [2024, '电子产品', '笔记本电脑', '在售', 20000, 4.9, 300],
    [2023, '美容', '护肤套装', '在售', 7000, 4.4, 120],
    [2023, '家居与园艺', '园艺工具', '在售', 6500, 4.3, 130],
    [2024, '健康', '健身追踪器', '在售', 9500, 4.6, 190],
    [2023, '玩具', '动作人偶', '在售', 4800, 4.7, 95],
    [2024, '汽车', '汽车配件', '在售', 3200, 4.5, 65],
    [2023, '体育用品', '篮球', '在售', 7600, 4.8, 150],
    [2024, '办公用品', '笔记本', '在售', 11000, 4.4, 220]
]);

// 向单元格 B16 添加 FORMULATEXT 函数。
sheet3.setFormula(15, 1, "=FORMULATEXT(B17)");

// 添加 GROUPBY 函数，将函数名作为 LAMBDA 函数参数。
sheet3.setFormula(16, 1, "=GROUPBY(H16:I28,L16:L28,SUM)");
```

## 引用自定义函数

你可以定义自定义函数，并在公式函数中使用它们。

```javascript
// 向工作表添加值。
sheet4.setArray(0, 0, [
    [5, -3, 0],
    [23, -1, -4],
    [-13, 0, -12]
]);
sheet4.setValue(4, 0, "公式文本: ->");
sheet4.setValue(5, 0, "公式结果: ->");

// 设置单元格 B6:D8 的颜色。
sheet4.getRange('B6:D8').backColor('浅蓝色');

// 设置单元格 B6 的颜色。
sheet4.getCell(5, 1).backColor('黄色');

// 定义一个自定义函数，检查值是否小于 0。
function IsNegativeFunction() {
    this.name = "ISNEGATIVE";
    this.maxArgs = 1;
    this.minArgs = 1;
}
IsNegativeFunction.prototype = new GC.Spread.CalcEngine.Functions.Function();
IsNegativeFunction.prototype.evaluate = function (arg) {
    if (parseInt(arg) < 0) {
return true;
    }
    return false;
};

// 向工作簿添加自定义函数。
spread.addCustomFunction(new IsNegativeFunction());

// 添加 MAP 函数，将自定义函数名作为 LAMBDA 函数参数。
sheet4.setFormula(5, 1, "=MAP(A1:C3,ISNEGATIVE)");

// 向单元格 B5 添加 FORMULATEXT 函数。
sheet4.setFormula(4, 1, "=FORMULATEXT(B6)");
```

> **注意**：要启用此功能，你必须启用 **allowDynamicArray** 属性。
