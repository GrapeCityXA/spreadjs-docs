# io

你可以以多种格式导入和导出甘特图工作表（GanttSheet），例如 JSON、SJS 和 Excel 格式。SpreadJS 还支持打印以及导出为 PDF 格式。
为了支持甘特图工作表的输入输出（IO）功能（JSON 序列化/反序列化除外），有必要根据输入输出操作添加特定的脚本文件以供引用。例如：

* \<script src="scripts/gc.spread.sheets.tablesheet.x.x.x.min.js">\</script>
* \<script src="scripts/gc.spread.sheets.ganttsheet.x.x.x.min.js">\</script>
* \<script src="scripts/gc.spread.sheets.print.x.x.x.min.js">\</script>
* \<script src="scripts/gc.spread.sheets.pdf.x.x.x.min.js">\</script>
* \<script src="scripts/gc.spread.sheets.io.x.x.x.min.js">\</script>

## 打开和导入

对于甘特图工作表的 JSON 序列化和反序列化，你需要使用 **Workbook** 类的 **toJSON** 和 **fromJSON** 方法。甘特图工作表的 JSON 数据存储在 **workbookJSONData.sheetTabs** 中，甘特图工作表的指定属性存储在 **ganttSheetTabData.gantt** 中。此外，在 **fromJSON** 过程中，**sheetType** 属性对于区分表格工作表（TableSheet）和甘特图工作表（GanttSheet）是必需的。
你可以打开和导入不同的文件类型。要打开 SJS 文件类型，使用 `spread.open()` 方法并传入 **openOptions**。如果文件类型是 SSJson 或 Excel，则使用 `spread.import()` 方法，并分别传入 **importSSJsonOptions** 或 **importXlsxOptions** 进行导入。
以下代码展示了如何打开和导入文件到 Spread 控件中。

```javascript
function openFile(e) {
    var fileNameItem = document.getElementById("file-name");
    var fileInput = document.getElementById("file-input");
    fileNameItem.value = 'Loading file...';
    var file = fileInput.files[0];
    if (!file) {
        alert("Upload a file first.");
        return;
    }

    const fileName = file.name;
    const extensionName = fileName.substring(fileName.lastIndexOf(".") + 1);

    // 定义打开 SJS 文件的选项。
    const openOptions = {
        "openMode": GC.Spread.Sheets.OpenMode.normal,
        "includeStyles": true,
        "includeFormulas": true,
        "fullRecalc": false,
        "dynamicReferences": true,
        "calcOnDemand": false,
        "includeUnusedStyles": true
    }

    // 定义导入 SSJson 文件的选项。
    const importSSJsonOptions = {
        "includeStyles": true,
        "includeFormulas": true,
        "frozenColumnsAsRowHeaders": false,
        "frozenRowsAsColumnHeaders": false,
        "fullRecalc": false,
        "incrementalLoading": false
    }

    // 定义导入 Excel 文件的选项。
    const importXlsxOptions = {
        "openMode": GC.Spread.Sheets.OpenMode.normal,
        "includeStyles": true,
        "includeFormulas": true,
        "frozenColumnsAsRowHeaders": false,
        "frozenRowsAsColumnHeaders": false,
        "fullRecalc": false,
        "dynamicReferences": true,
        "calcOnDemand": false,
        "includeUnusedStyles": true,
        "password": ""
    }

    // 导入 SJS 文件类型。
    if (extensionName === 'sjs') {
        spread.open(file, function () {
            fileNameItem.value = fileName;
            fileInput.value = '';
        }, function () { }, openOptions);
    }

    // 导入 SSJson 文件类型。
    else if (extensionName === 'ssjson') {
        spread.import(file, function () {
            fileNameItem.value = fileName;
            fileInput.value = '';
        }, function () { }, importSSJsonOptions);
    }

    // 导入 Excel 文件类型。
    else if (extensionName === 'xlsx') {
        spread.import(file, function () {
            fileNameItem.value = fileName;
            fileInput.value = '';
        }, function () { }, importXlsxOptions);
    }
}
```

## 保存和导出

要将甘特图工作表数据保存为 SJS 格式，你需要使用 `spread.save()` 方法，并指定必要的 **saveOptions**。如果文件类型是 SSJson 或 Excel，则使用 `spread.export()` 方法，并分别传入 **exportSSJsonOptions** 或 **exportXlsxOptions** 进行导出。

> **注意事项**：
>
> * 确保在导出时将序列化选项 **includeBindingSource** 设置为 true。
> * 导出时，甘特图工作表的数据列在 Excel 中作为表格导出，甘特图列作为图片导出。

以下代码展示了如何从 Spread 控件保存和导出文件：

```javascript
function exportFile(fileName) {
    const extensionName = fileName.substring(fileName.lastIndexOf(".") + 1);

    // 定义导出 SSJson 文件的选项。
    const exportSSJsonOptions = {
        "includeBindingSource": true,
        "includeStyles": true,
        "includeFormulas": true,
        "saveAsView": false,
        "rowHeadersAsFrozenColumns": false,
        "columnHeadersAsFrozenRows": false,
        "includeAutoMergedCells": false,
        "fileType": GC.Spread.Sheets.FileType.ssjson
    }

    // 定义保存 SJS 文件的选项。
    const saveOptions = {
        "includeBindingSource": true,
        "includeStyles": true,
        "includeFormulas": true,
        "saveAsView": false,
        "includeAutoMergedCells": false,
        "includeCalcModelCache": false,
        "includeUnusedNames": true,
        "includeEmptyRegionCells": true
    }

    // 定义导出 Excel 文件的选项。
    const exportXlsxOptions = {
        "includeBindingSource": true,
        "includeStyles": true,
        "includeFormulas": true,
        "saveAsView": false,
        "rowHeadersAsFrozenColumns": false,
        "columnHeadersAsFrozenRows": false,
        "includeAutoMergedCells": false,
        "includeCalcModelCache": false,
        "includeUnusedNames": true,
        "includeEmptyRegionCells": true,
        "fileType": GC.Spread.Sheets.FileType.excel
    }

    // 导出 SJS 文件类型。
    if (extensionName === 'sjs') {
        spread.save(function (blob) { saveAs(blob, fileName); }, function () { }, saveOptions);
    }

    // 导出 SSJson 文件类型。
    else if (extensionName === 'ssjson') {
        spread.export(function (blob) { saveAs(blob, fileName); }, function () { }, exportSSJsonOptions);
    }

    // 导出 Excel 文件类型。
    else if (extensionName === 'xlsx') {
        spread.export(function (blob) { saveAs(blob, fileName); }, function () { }, exportXlsxOptions);
    }
}
```

## 保存为 PDF

使用工作簿（workbook）的 `savePDF()` 方法将甘特图工作表导出为 PDF 格式。
以下代码展示了如何从 Spread 控件导出 PDF 文件：

```javascript
function ExportToPDF(spread) {
            spread.savePDF(function (blob) {
                var url = URL.createObjectURL(blob);
                pwin = window.open(url);
            });
}
```

## 打印甘特图工作表

使用 **Workbook** 类的 `print()` 方法来打印甘特图工作表。一般来说，甘特图列较大，无法在一页上打印完。因此，它会打印在最后一个普通列附近。
以下代码展示了如何从 Spread 控件打印甘特图工作表：

```javascript
function functionPrint(spread) {
    spread.print();
}
```
