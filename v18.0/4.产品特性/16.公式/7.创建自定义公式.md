# formula-customize

如果你经常使用一些不在内置函数中的函数，或者你希望将一些内置函数组合成一个单一的函数，那么你可以通过定义自己的自定义函数来实现。你可以像调用任何内置函数一样调用这些自定义函数。
这些自定义函数可以返回指定的值或数组。以下部分将对此进行说明：

## 自定义函数中返回单个值

以下代码示例创建了一个返回值的自定义函数。

```javascript
var spread = new GC.Spread.Sheets.Workbook(document.getElementById("ss"),{sheetCount:3});
var activeSheet = spread.getActiveSheet();
// 添加自定义函数
// 在单元格中输入 =myfunc(1)
// 查看结果
function myfunc() {}
myfunc.prototype = new GC.Spread.CalcEngine.Functions.Function("myfunc", 0, 0, {name: "myfunc",description: "这是我的第一个函数"});
myfunc.prototype.evaluate = function (args) {
return 100;}
spread.addCustomFunction(new myfunc());
```

## 自定义函数中返回数组

以下代码示例创建了一个返回数组的自定义函数。

```javascript
$(document).ready(function () {
    // 初始化 Spread
    var spread = new GC.Spread.Sheets.Workbook(document.getElementById('ss'), { sheetCount: 1 });

    // 获取活动工作表
    var sheet = spread.getSheet(0);

    // 允许动态数组
    spread.options.allowDynamicArray = true;

    // 创建一个阶乘函数
    function FactorialFunction() {
        this.name = "FACTORIAL";
        this.maxArgs = 1;
        this.minArgs = 1;
    }
    FactorialFunction.prototype = new GC.Spread.CalcEngine.Functions.Function();

    // 计算公式的结果
    FactorialFunction.prototype.evaluate = function (arg) {
        var t = 1;
        var result = [];
        if (arguments.length === 1 &&!isNaN(parseInt(arg))) {
            for (var i = 1; i <= arg; i++) {
                t = i * t;
                result.push(t);
            }

            // 返回 CALCArray 对象作为函数的结果
            return new GC.Spread.CalcEngine.CalcArray([result]);
        }
        return "#VALUE!";
    };

    // 为自定义函数添加描述
    FactorialFunction.prototype.description = function () {
        return {
            name: "FACTORIAL",
            description:
                "这是一个计算从 1 开始的阶乘并在一行的不同单元格中显示的函数",
            parameters: [
			   { name: "number" }
            ]
        };
    };
    var factorial = new FactorialFunction();
    sheet.addCustomFunction(factorial);

    // 合并以单元格(0,0)为起点的三列
    sheet.addSpan(0, 0, 1, 7, GC.Spread.Sheets.SheetArea.viewport);
    sheet.setValue(0, 0, "这个自定义阶乘函数计算从 1 到 7 的阶乘 ");
    for (var i = 1; i < 8; i++) {
        sheet.setText(2, i - 1, i + "!");
    }

    // 设置自定义公式
    sheet.setFormula(3, 0, "=Factorial(7)");
    var style = new GC.Spread.Sheets.Style();
    style.font = "bold 12px Arial";
    style.foreColor = "white";
    style.backColor = "#5B9BD5";
    style.hAlign = GC.Spread.Sheets.HorizontalAlign.center;
    style.vAlign = GC.Spread.Sheets.VerticalAlign.center;
    sheet.setStyle(0, 0, style, GC.Spread.Sheets.SheetArea.viewport);
});
```
