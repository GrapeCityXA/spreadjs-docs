# using-celltypes-in-grouping

在SpreadJS中，你可以向分组列添加不同的单元格类型。例如，你可以向分组列添加“超链接”单元格类型，让用户点击超链接单元格类型以导航到特定页面。
![超链接_TS分组](/DOCUMENT_SITE_LINK_PREFIX_HERE/document-site-files/images/5fefd5d8-238b-4c13-893b-6f22fcc5b9dd/Hyperlink_TSGrouping.0f1ab6.b845c3.png)

> **注意**：单元格类型仅可应用于[表格分组布局](gcdocsite__documentlink?toc-item-id=91cdc22b-1a69-43e1-935b-b1ddee4cfe37)下的分组列。
> 要在分组列中创建超链接，需定义一个“超链接”单元格类型并将其分配给该列。以下示例代码添加了使用超链接单元格类型的分组信息，并禁用了按值或列表进行排序和筛选的功能。

```javascript
var spread, sheet;
$(document).ready(function () {
    // 初始化Spread。
    spread = new GC.Spread.Sheets.Workbook(document.getElementById('ss'), { sheetCount: 0 });
    spread.options.autoFitType = GC.Spread.Sheets.AutoFitType.cellWithHeader;
    // 初始化数据管理器。
    var dataManager = spread.dataManager();
    var myTable = dataManager.addTable("myTable", {
        remote: {
            read: {
                url: 'https://demodata.grapecity.com/wwi/api/v1/stockItems'
            }
        },
        schema: {
            columns: {
                validFrom: { dataType: "date" }
            }
        }
    });
    // 初始化表格工作表。
    sheet = spread.addSheetTab(0, "TableSheet1", GC.Spread.Sheets.SheetType.tableSheet);
    h = new GC.Spread.Sheets.CellTypes.HyperLink();
    h.linkToolTip('这是指向SpreadJS网站的链接');
    spread.commandManager().register('setSheetTabStyle', {
        canUndo: true,
        execute: function (context, options, isUndo) {
           alert("已点击");
        }
    }, null, false, false, false, false);
    h.onClickAction(() => {
        spread.commandManager().execute({ cmd: 'setSheetTabStyle' });
    });
    // 将视图绑定到表格工作表。
    myTable.fetch().then(function () {
        var view = myTable.addView("myView", [
            { value: "stockItemKey", width: 150, caption: "库存商品键" },
            { value: "stockItem", width: 120, caption: "库存商品" },
            { value: "buyingPackage", width: 120, caption: "采购包装" },
            { value: "sellingPackage", width: 120, caption: "销售包装" },
            { value: "unitPrice", width: 150, caption: "单价" },
            { value: "taxRate", width: 120, caption: "税率" },
            { value: "validFrom", width: 200, caption: "生效日期" }
        ]);
        sheet.setDataView(view);
    });
});
function grouping() {
    // 通过单个字段（销售包装）创建分组依据，包含汇总字段和切片字段。
    sheet.groupBy([
        {
            caption: "销售包装", field: "sellingPackage", width: 150, style: { backColor: "#F9CA9A" }, style: { cellType: h },
            allowSort: false,
            allowFilterByValue: false,
            allowFilterByList: false,
            summaryFields: [
                {
                    caption: "SUM(单价)",
                    formula: "=SUM([unitPrice])",
                    slice: { field: "=YEAR([@validFrom])", width: 120, style: { backColor: "#FCE3CA", formatter: "#,##0.00" } },
                    width: 150,
                    style: { backColor: "#FAD7B2", formatter: "#,##0.00" }
                }
            ]
        }
    ]);
}
```
