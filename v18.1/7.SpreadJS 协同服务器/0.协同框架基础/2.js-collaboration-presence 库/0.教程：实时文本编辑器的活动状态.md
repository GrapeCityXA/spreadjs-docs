# real-time-text-editor

本教程在[“实时协同文本编辑器教程”](gcdocsite__documentlink?toc-item-id=586c843b-82b8-4fd4-8bc5-276ac583ce1b)基础上添加Presence功能，用于展示在线用户及其光标位置。用户可在编辑器中实时查看他人状态（如用户名、选区位置等），提升协同体验。

## 效果预览

当一位用户在编辑器中输入字符时，其他用户可看到其光标位置、选中区域、用户名及输入内容。

## 前提条件

* 熟悉JavaScript和HTML/CSS基础知识

## 步骤1：安装依赖包

在项目根目录运行以下命令安装Presence相关的npm包：

```bash
npm install @grapecity-software/js-collaboration-presence  @grapecity-software/js-collaboration-presence-client
npm install quill-cursors
```

## 步骤2：更新服务器以支持Presence功能

修改`server.js`文件，添加Presence功能：

```javascript
import express from "express";
import { createServer } from "http";
import { Server } from "@grapecity-software/js-collaboration";
import OT from "@grapecity-software/js-collaboration-ot";
import { presenceFeature } from "@grapecity-software/js-collaboration-presence";
import richText from "rich-text";

const app = express();
const httpServer = createServer(app);
const server = new Server({ httpServer });

// 注册富文本类型
OT.TypesManager.register(richText.type);

// 初始化OT文档服务
server.useFeature(OT.documentFeature());

// 初始化Presence服务
server.useFeature(presenceFeature());

// 服务静态文件
app.use(express.static("public"));

// 启动服务器
httpServer.listen(8080, () => {
  console.log("Server running at http://localhost:8080");
});
```

## 步骤3：更新客户端以支持Presence功能

修改客户端文件以添加在线用户光标显示功能，更新`public/client.js`：

```javascript
import { Client } from "@grapecity-software/js-collaboration-client";
import * as OT from "@grapecity-software/js-collaboration-ot-client";
import { Presence } from "@grapecity-software/js-collaboration-presence-client";
import richText from "rich-text";
import Quill from "quill";
import QuillCursors from "quill-cursors";

Quill.register('modules/cursors', QuillCursors);

// 注册富文本类型
OT.TypesManager.register(richText.type);

// 连接服务器并加入房间
const connection = new Client().connect("room-id");
const doc = new OT.SharedDoc(connection);
const quill = new Quill("#editor", { theme: "bubble", modules: { cursors: true } });
const cursors = quill.getModule('cursors');

// 初始化Presence
const presence = new Presence(connection);
const userId = `user-${Math.random().toString(36).substring(2, 9)}`; // 生成随机用户ID
const DEFAULT_COLOR_SCHEME = ['#0000ff', '#008000', '#9900cc', '#800000', '#00cc33', '#cc6600', '#cc0099'];

// 订阅文档
doc.subscribe().then(async () => {
    if (!doc.type) {
        try {
            await doc.create([{ insert: "Hi!" }], richText.type.uri, {});
        } catch (err) {
            console.error("Create Document Error:", err);
        }
    }
    quill.setContents(doc.data);
    quill.on("text-change", (delta, oldDelta, source) => {
        if (source !== "user") return;
        doc.submitOp(delta, { source: connection.id });
    });
    doc.on("op", (op, source) => {
        if (source === connection.id) return;
        quill.updateContents(op);

        // 更新内容会导致选区变化，但Quill不会触发selection-change事件
        presence.submitLocalStateField('selection', quill.getSelection());
    });
});

// 显示远程用户状态
function updatePresenceList() {
    cursors.clearCursors();

    for (const id in presence.otherStates) {
        const data = presence.otherStates[id];
        cursors.createCursor(id, data.userId, data.color);
        cursors.moveCursor(id, data.selection);
    }
}


// 订阅Presence更新
presence.subscribe().then(() => {
    updatePresenceList();

    let color = DEFAULT_COLOR_SCHEME[Object.keys(presence.otherStates).length % DEFAULT_COLOR_SCHEME.length];

    presence.submitLocalState({ userId, selection: quill.getSelection(), color });

    // 选区变化时更新本地Presence状态
    quill.on("selection-change", (range) => {
        presence.submitLocalStateField('selection', range);
    });

    presence.on("add", () => updatePresenceList());
    presence.on("update", () => updatePresenceList());
    presence.on("remove", () => updatePresenceList());
})

// 错误处理
connection.on("error", (err) => console.error("Connection error:", err.message));
doc.on("error", (err) => console.error("Document error:", err.message));
```

## 步骤4：运行与测试

1. 打包客户端代码

    ```bash
    npm run build
    ```
2. 启动服务器

    ```bash
    npm run start
    ```

    控制台应显示：`Server running at http://localhost:8080.`
3. 测试功能
    1. 打开浏览器访问 *http://localhost:8080*。
    2. 在多个窗口中打开相同地址，编辑内容并观察实时同步效果。
