# add-elements-to-a-tab

SpreadJS Designer 组件允许您在功能区（Ribbon）组件的选项卡中添加各种元素，如按钮、复选框和下拉菜单。这使您能够根据需要自定义功能区，以实现更实用和高效的使用。
要执行这些任务，您需要遵循以下步骤：

1. 初始化一个工作簿并访问活动工作表。
2. 访问 SpreadJS Designer 组件的默认配置设置。
3. 为新元素定义布局结构和属性。
4. 将元素添加到选项卡中。
5. 创建与新元素关联的命令。
6. 使用新配置的设置初始化设计器实例。

下面我们通过一些示例来详细讨论。

## 添加按钮

以下代码示例展示了如何在“HOME”选项卡中添加一个名为“Save Data”（保存数据）的按钮。

```javascript
// 访问默认配置
var config = GC.Spread.Sheets.Designer.DefaultConfig;

// 设置新按钮的布局
var saveData = {
    label: 'Save Data',
    thumbnailClass: '',
    commandGroup: {
        children: [
            {
                direction: 'vertical',
                commands: ['cmdSaveData'],
            },
        ],
    },
};

// 将新按钮添加到配置的功能区选项卡中
config.ribbon[0].buttonGroups.unshift(saveData);

// 为新按钮创建命令
config.commandMap = {
    cmdSaveData: {
        title: 'Save data to server',
        text: 'Save',
        iconClass: 'cmdSaveData',
        bigButton: 'true',
        commandName: 'cmdSaveData',
        execute: async (context, propertyName, fontItalicChecked) => {
            // 自定义操作
            alert('Save data successfully.');
        },
    },
};

// 初始化设计器实例
var designer = new GC.Spread.Sheets.Designer.Designer(document.getElementById('designerHost'));
designer.setConfig(config);
```

此外，如果您愿意，还可以为新按钮设置样式，例如背景图像。

```html
<!-- 新按钮的 CSS 样式设置 -->
<style>
.cmdSaveData {
    background-image: url('./saveData.png');
    background-size: 35px 35px;
}
</style>
```

## 添加复选框

您可以向选项卡中添加复选框，并且还可以切换复选框的选中或未选中状态。
以下代码示例展示了如何在“HOME”选项卡中添加一个名为“My Checkbox”（我的复选框）的自定义复选框。

```javascript
// 访问默认配置
var config = GC.Spread.Sheets.Designer.DefaultConfig;

// 设置新复选框的布局
var NewCheckBox = {
    label: 'SpreadJS',
    thumbnailClass: '',
    commandGroup: {
        children: [
            {
                direction: 'vertical',
                commands: ['MyCheckbox'],
            },
        ],
    },
};

// 将新复选框添加到配置的功能区选项卡中
config.ribbon[0].buttonGroups.unshift(NewCheckBox);

// 为新复选框创建命令
config.commandMap = {
    MyCheckbox: {
        title: 'Custom Checkbox',
        text: 'My Checkbox',
        type: 'checkbox',
        commandName: 'MyCheckbox',
        execute: async (context, propertyName) => {
            let state = context.getData('myCheckState');
            context.setData('myCheckState', !state);
        },
        getState: (context) => {
            return context.getData('myCheckState');
        },
    },
};

// 初始化设计器实例
var designer = new GC.Spread.Sheets.Designer.Designer(document.getElementById('designerHost'));
designer.setConfig(config);
```

## 添加下拉菜单

您可以向功能区选项卡中添加可编辑和不可编辑的下拉菜单。
以下代码示例描述了如何在“HOME”选项卡中添加下拉菜单。

```javascript
// 访问默认配置
var config = GC.Spread.Sheets.Designer.DefaultConfig;

// 创建一个可编辑的下拉菜单并定义命令对象
const customDropDownList1 = [
    { text: '10', value: 10 },
    { text: '20', value: 20 },
    { text: '30', value: 30 },
    { text: '40', value: 40 },
    { text: '50', value: 50 },
];
const customCommand1 = {
    title: 'Custom Command 1',
    type: 'editableSelect',
    width: 70,
    commandName: 'customCommand1',
    enableContext: 'true',
    dropdownList: customDropDownList1,
    execute: function (context, value) {
        let spreadObj = context.getWorkbook();
        let activeSheet = spreadObj.getActiveSheet();
        let cellTag = activeSheet.getTag(activeSheet.getActiveRowIndex(), activeSheet.getActiveColumnIndex());
        let newCellTag = {
            ...cellTag,
            customCommand1: value,
        };
        activeSheet.setTag(activeSheet.getActiveRowIndex(), activeSheet.getActiveColumnIndex(), newCellTag);
    },
    getState: function (context) {
        let spreadObj = context.getWorkbook();
        let activeSheet = spreadObj.getActiveSheet();
        let cellTag = activeSheet.getTag(activeSheet.getActiveRowIndex(), activeSheet.getActiveColumnIndex());
        if (cellTag && cellTag.customCommand1) {
            const nearestValue = findNearestValue(cellTag.customCommand1, customDropDownList1);
            return nearestValue.value;
        }
        return customDropDownList1[0].value;
    },
};

// 创建一个不可编辑的下拉菜单并定义命令对象
const customDropDownList2 = [
    { text: 'List 1', value: 'value1' },
    { text: 'List 2', value: 'value2' },
    { text: 'List 3', value: 'value3' },
    { text: 'List 4', value: 'value4' },
];
const customCommand2 = {
    title: 'Custom Command 2',
    type: 'comboBox',
    text: 'Custom Command 2',
    comboWidth: 70,
    commandName: 'customCommand2',
    dropdownMaxHeight: 500,
    dropdownList: customDropDownList2,
    execute: function (context, value) {
        let spreadObj = context.getWorkbook();
        let activeSheet = spreadObj.getActiveSheet();
        let cellTag = activeSheet.getTag(activeSheet.getActiveRowIndex(), activeSheet.getActiveColumnIndex());
        let newCellTag = {
            ...cellTag,
            customCommand2: value,
        };
        activeSheet.setTag(activeSheet.getActiveRowIndex(), activeSheet.getActiveColumnIndex(), newCellTag);
    },
    getState: function (context) {
        let spreadObj = context.getWorkbook();
        let activeSheet = spreadObj.getActiveSheet();
        let cellTag = activeSheet.getTag(activeSheet.getActiveRowIndex(), activeSheet.getActiveColumnIndex());
        if (cellTag && cellTag.customCommand2) {
            return cellTag.customCommand2;
        }
        return customDropDownList2[0].value;
    },
};

// 设置新下拉菜单的布局
var newDropDowns = {
    label: 'DropDowns',
    commandGroup: {
        children: [
            {
                direction: 'vertical',
                commands: ['customCommand1', 'customCommand2'],
            },
        ],
    },
};

// 将新下拉菜单添加到 HOME 选项卡
config.ribbon[0].buttonGroups.unshift(newDropDowns);

// 为新下拉菜单创建命令
config.commandMap = {
    customCommand1: customCommand1,
    customCommand2: customCommand2,
};

// 根据用户输入从下拉列表中确定最接近的值
function findNearestValue(value, customDropDownList) {
    return customDropDownList.reduce((prev, curr) => (Math.abs(curr.value - value) < Math.abs(prev.value - value) ? curr : prev));
}

// 初始化设计器实例
var designer = new GC.Spread.Sheets.Designer.Designer(document.getElementById('designerHost'));
designer.setConfig(config);
```
